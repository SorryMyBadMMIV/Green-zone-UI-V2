> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the gisda_03 grid 600 × 600 cells of about 5 m (was 30 × 30 of 100 m). The grid is drawn as a fast image instead of 360,000 polygons, and the mock data and zones are regenerated at 5 m.

**Architecture:**
- `classifyGrid` turns the NDVI grid into one byte per cell, and `classAreas` sums real cell areas from the corner lattice.
- `ClassifiedGridLayer` (a Leaflet `GridLayer`) paints a 1-pixel-per-cell image into map tiles, using two affine triangles per tile so the 4 GISTDA corners land exactly.
- Mock data is regenerated by rasterising OpenStreetMap shapes onto the 5 m grid.

**Tech Stack:** unchanged (Vite 8, React 19, Tailwind 4, Leaflet 1.9 + react-leaflet 5, Vitest 5).

```js
import { GREEN_NDVI, WATER_NDVI } from './constants.js';
import { pointInFrame, ringAreaM2 } from './geo.js';

/** One byte per cell: the class codes of the classified grid. */
export const CLASS = { unknown: 0, water: 1, grey: 2, green: 3 };

/** Decides a cell from its NDVI: 'water' (< 0), 'grey' (< 0.3) or 'green'. null when NDVI is missing. */
export function classifyNdvi(ndvi) {
  if (typeof ndvi !== 'number' || !Number.isFinite(ndvi)) return null;
  if (ndvi < WATER_NDVI) return 'water';
  return ndvi >= GREEN_NDVI ? 'green' : 'grey';
}

/**
 * Decides every cell of an NDVI grid ({ rows, cols, ndvi[row][col] }, row 0 = north edge).
 * Returns { rows, cols, classes } with one CLASS code per cell, row by row.
 */
export function classifyGrid(grid) {
  const { rows, cols } = grid;
  const classes = new Uint8Array(rows * cols);
  for (let row = 0; row < rows; row++) {
    const values = grid.ndvi[row] ?? [];
    for (let col = 0; col < cols; col++) {
      const cls = classifyNdvi(values[col]);
      classes[row * cols + col] = cls ? CLASS[cls] : CLASS.unknown;
    }
  }
  return { rows, cols, classes };
}

/**
 * Total area in m² of each class. Every cell's four corners are computed from the frame's corner
 * lat/lngs (the same bilinear split as the map), so the classes always add up to the frame's area.
 */
export function classAreas(frame, { rows, cols, classes }) {
  const stride = cols + 1;
  const corners = new Array((rows + 1) * stride);
  for (let row = 0; row <= rows; row++) {
    for (let col = 0; col <= cols; col++) corners[row * stride + col] = pointInFrame(frame.corners, col / cols, row / rows);
  }
  const totals = [0, 0, 0, 0];
  const ring = new Array(5);
  for (let row = 0; row < rows; row++) {
    for (let col = 0; col < cols; col++) {
      const topLeft = row * stride + col;
      ring[0] = corners[topLeft];
      ring[1] = corners[topLeft + stride];
      ring[2] = corners[topLeft + stride + 1];
      ring[3] = corners[topLeft + 1];
      ring[4] = corners[topLeft];
      totals[classes[row * cols + col]] += ringAreaM2(ring);
    }
  }
  return { unknown: totals[CLASS.unknown], water: totals[CLASS.water], grey: totals[CLASS.grey], green: totals[CLASS.green] };
}
```

```js
/**
 * The canvas transform [a, b, c, d, e, f] of the affine map that sends three source points onto
 * three destination points: x' = a·x + c·y + e, y' = b·x + d·y + f (the order setTransform takes).
 */
export function affineFromPoints([[x0, y0], [x1, y1], [x2, y2]], [[u0, v0], [u1, v1], [u2, v2]]) {
  const det = (x1 - x0) * (y2 - y0) - (x2 - x0) * (y1 - y0);
  const a = ((u1 - u0) * (y2 - y0) - (u2 - u0) * (y1 - y0)) / det;
  const c = ((u2 - u0) * (x1 - x0) - (u1 - u0) * (x2 - x0)) / det;
  const b = ((v1 - v0) * (y2 - y0) - (v2 - v0) * (y1 - y0)) / det;
  const d = ((v2 - v0) * (x1 - x0) - (v1 - v0) * (x2 - x0)) / det;
  return [a, b, c, d, u0 - a * x0 - c * y0, v0 - b * x0 - d * y0];
}

/** "#22c55e" → [34, 197, 94] */
export function hexToRgb(hex) {
  const n = Number.parseInt(hex.slice(1), 16);
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
}

/** RGBA pixels (4 bytes per cell, row by row) for the class codes, with palette[code] = [r, g, b, a]. */
export function paintClasses({ rows, cols, classes }, palette) {
  const pixels = new Uint8ClampedArray(rows * cols * 4);
  for (let i = 0; i < rows * cols; i++) pixels.set(palette[classes[i]], i * 4);
  return pixels;
}
```

```jsx
import { useEffect } from 'react';
import { useMap } from 'react-leaflet';
import L from 'leaflet';
import { COLORS } from '../../lib/constants.js';
import { CLASS } from '../../lib/grid.js';
import { affineFromPoints, hexToRgb, paintClasses } from '../../lib/raster.js';

const ALPHA = 120; // ≈47% opacity
const PALETTE = [];
PALETTE[CLASS.unknown] = [0, 0, 0, 0];
PALETTE[CLASS.water] = [0, 0, 0, 0]; // not drawn: the basemap already shows the water
PALETTE[CLASS.grey] = [...hexToRgb(COLORS.gridGrey), ALPHA];
PALETTE[CLASS.green] = [...hexToRgb(COLORS.gridGreen), ALPHA];

/** The class image (one pixel per cell) plus halved copies, used when the map is zoomed out. */
function buildMipmaps(classified) {
  const base = document.createElement('canvas');
  base.width = classified.cols;
  base.height = classified.rows;
  const pixels = paintClasses(classified, PALETTE);
  base.getContext('2d').putImageData(new ImageData(pixels, classified.cols, classified.rows), 0, 0);
  const levels = [base];
  while (levels.at(-1).width > 32 && levels.at(-1).height > 32) {
    const previous = levels.at(-1);
    const next = document.createElement('canvas');
    next.width = Math.ceil(previous.width / 2);
    next.height = Math.ceil(previous.height / 2);
    next.getContext('2d').drawImage(previous, 0, 0, next.width, next.height);
    levels.push(next);
  }
  return levels;
}

/** Draws `image` mapped from the source triangle onto the tile triangle, grown by 1 px so neighbours meet without a seam. */
function drawTriangle(ctx, image, source, target) {
  const cx = (target[0][0] + target[1][0] + target[2][0]) / 3;
  const cy = (target[0][1] + target[1][1] + target[2][1]) / 3;
  ctx.save();
  ctx.beginPath();
  target.forEach(([x, y], i) => {
    const length = Math.hypot(x - cx, y - cy) || 1;
    const px = x + (x - cx) / length;
    const py = y + (y - cy) / length;
    if (i === 0) ctx.moveTo(px, py);
    else ctx.lineTo(px, py);
  });
  ctx.closePath();
  ctx.clip();
  ctx.setTransform(...affineFromPoints(source, target));
  ctx.drawImage(image, 0, 0);
  ctx.restore();
}

/** A Leaflet GridLayer whose tiles show the classified frame (two affine triangles per tile). */
function createGridLayer(frame, classified, pane) {
  const mipmaps = buildMipmaps(classified);
  const { topLeft, topRight, bottomRight, bottomLeft } = frame.corners;
  const corners = [topLeft, topRight, bottomRight, bottomLeft].map(([lat, lng]) => L.latLng(lat, lng));

  const FrameGrid = L.GridLayer.extend({
    createTile(coords) {
      const size = this.getTileSize();
      const tile = document.createElement('canvas');
      tile.width = size.x;
      tile.height = size.y;
      const origin = coords.scaleBy(size);
      const [tl, tr, br, bl] = corners.map((latLng) => {
        const point = this._map.project(latLng, coords.z).subtract(origin);
        return [point.x, point.y];
      });
      // Use the copy whose pixels are closest to the on-screen cell size: smooth when zoomed out,
      // crisp squares (one per cell) when zoomed in.
      const pixelsPerCell = Math.hypot(tr[0] - tl[0], tr[1] - tl[1]) / classified.cols;
      const level = Math.max(0, Math.min(mipmaps.length - 1, Math.round(Math.log2(1 / pixelsPerCell))));
      const image = mipmaps[level];
      const ctx = tile.getContext('2d');
      ctx.imageSmoothingEnabled = level > 0 || pixelsPerCell < 1;
      const [w, h] = [image.width, image.height];
      // Two triangles, each with its own affine map, so all four frame corners land exactly.
      drawTriangle(ctx, image, [[0, 0], [w, 0], [0, h]], [tl, tr, bl]);
      drawTriangle(ctx, image, [[w, 0], [w, h], [0, h]], [tr, br, bl]);
      return tile;
    },
  });

  return new FrameGrid({ pane, bounds: L.latLngBounds(corners), updateWhenZooming: false });
}

/** The classified grid of a frame (green / grey cells), drawn from a one-pixel-per-cell image. */
export default function ClassifiedGridLayer({ frame, classified, pane }) {
  const map = useMap();
  useEffect(() => {
    const layer = createGridLayer(frame, classified, pane);
    layer.addTo(map);
    return () => {
      layer.remove();
    };
  }, [map, frame, classified, pane]);
  return null;
}
```
